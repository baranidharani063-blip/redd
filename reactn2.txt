2. _layout.tsx

இது Navigation setup.

import { Stack } from "expo-router";


export default function RootLayout() {
  return (
    <Stack>
      <Stack.Screen
        name="index"
        options={{
          title: "Home",
        }}
      />


      <Stack.Screen
        name="details"
        options={{
          title: "Details",
        }}
      />


      <Stack.Screen
        name="profile"
        options={{
          title: "Profile",
        }}
      />
    </Stack>
  );
}
இதன் meaning:
Stack
 │
 ├── Home
 ├── Details
 └── Profile

இது Stack Navigation.

3. index.tsx

இது நம்ம Home Screen.

import { View, Text, Button, StyleSheet } from "react-native";
import { useRouter } from "expo-router";


export default function HomeScreen() {
  const router = useRouter();


  return (
    <View style={styles.container}>


      <Text style={styles.title}>
        🏠 Home Screen
      </Text>


      <Button
        title="Go to Details"
        onPress={() => router.push("/details")}
      />


      <View style={styles.space} />


      <Button
        title="Go to Profile"
        onPress={() => router.push("/profile")}
      />


    </View>
  );
}


const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: "center",
    alignItems: "center",
    padding: 20,
  },


  title: {
    fontSize: 30,
    fontWeight: "bold",
    marginBottom: 30,
  },


  space: {
    height: 15,
  },
});

🧠 முக்கியமான 4 lines

அதாவது:

router.push()
     ↓
Next Screen


router.back()
     ↓
Previous Screen

   ↓
/

   ↓
/details


profile.tsx
   ↓
/profile
